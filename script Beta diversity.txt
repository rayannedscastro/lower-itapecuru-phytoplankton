install.packages("tidyverse")

library(tidyverse)
library(betapart)
library(wesanderson)
library(reshape2)

#Carregar dados


biologico <- DADOS
View(biologico)

######################DIVERSIDADE BETA TOTAL ###############################

#Calcular a Diversidade Beta Total  - para calcular a betadiversidade total podemos usar a 
#fun??o beta.multi do pacote betapart

# Use esse c?digo caso seus dados sejam de abund?ncia.

biologico.trans <- ifelse(biologico > 0, 1, 0)
View(biologico.trans)

beta_total <- beta.multi(biologico.trans,
                         index.family = "sorensen")
beta_total

##################### DIVERSIDADE BETA PAR-A-PAR #######################################
#Beta diversidade par-a-par  - Para calcular a betadiversidade para cada par de localidade podemos
#usar a fun??o beta.multi do pacote betapart

(beta_par <- beta.pair(biologico.trans,
                      index.family = "sorensen"))

#Grafico da diversidade beta 

get_upper_tri <- function(cormat){
  cormat[lower.tri(cormat)]<- NA
  return(cormat)
}

sim <- get_upper_tri(as.matrix(beta_par$beta.sim))
summary(sim) 

sor <- get_upper_tri(as.matrix(beta_par$beta.sor))
summary(sor) 

sne <- get_upper_tri(as.matrix(beta_par$beta.sne))
summary(sne) 

melted_cormat <- reshape2::melt(sim)
melted_cormat2 <- reshape2::melt(sor)
melted_cormat3 <- reshape2::melt(sne)

melted_cormat <- rbind(melted_cormat2, melted_cormat, melted_cormat3)

melted_cormat$metric <- factor(rep(c("Total", "Turnover", "Nestedness"), 
                                   each = nrow(melted_cormat2)), 
                               c("Total", "Turnover", "Nestedness"))

pal <- wes_palette("Zissou1", 100, type = "continuous")

g <- ggplot(data = melted_cormat, aes(Var2, Var1, fill = value))+
  geom_tile(color = "white")+
  scale_fill_gradientn(colours = pal, 
                       name="Beta diversity") +
  theme_minimal()+ 
  theme(axis.text.x = element_text(angle = 90, vjust = 0.5, hjust = 0),
        text = element_text(size = 12)) +
  coord_fixed() +
  xlab("") +
  ylab("") +
  facet_wrap(metric ~.) +
  ggtitle("Beta diversity")
g
